	input [511:0] data,
	output reg [511:0] q);

	always @(posedge clk) begin
		if (load) begin
			q <= data;
		end else begin
			integer i;
			for (i = 0; i < 512; i = i + 1) begin
				if (i == 0) begin
					q[i] <= 1'b0 ^ q[i+1]; // Left boundary is 0
				end else if (i == 511) begin
					q[i] <= q[i-1] ^ 1'b0; // Right boundary is 0
				end else begin
					q[i] <= q[i-1] ^ q[i+1];
				end
			end
		end
	end
