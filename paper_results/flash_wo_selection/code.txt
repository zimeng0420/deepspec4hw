def web_research(state: WebSearchState, config: RunnableConfig) -> OverallState:
    logger= logging.getLogger(f"task_{state.get('task_id', 'unknown')}")
    logger.info("Starting web research")
    """LangGraph node that performs web research using the native Google Search API tool.

    Executes a web search using the native Google Search API tool in combination with Gemini 2.0 Flash.

    Args:
        state: Current graph state containing the search query and research loop count
        config: Configuration for the runnable, including search API settings

    Returns:
        Dictionary with state update, including sources_gathered, research_loop_count, and web_research_results
    """
    # Configure
    configurable = Configuration.from_runnable_config(config)

    formatted_prompt = web_searcher_instructions.format(
        current_date=get_current_date(),
        query=state["search_query"],
        research_topic=get_research_topic(state["messages"]),
    )


    # Uses the google genai client as the langchain client doesn't return grounding metadata
    try:
        # Search with the Google GenAI client
        response = genai_client.models.generate_content(
            model=configurable.answer_model,
            contents=formatted_prompt,
            config={
                "tools": [{"google_search": {}}],
                "temperature": 0,
            },
        )
        print(f"*************************************")
        print(f"Using answer model: {configurable.answer_model}")
      
        # Check whether the response is valid
        if not response or not response.text or response.text.strip() == "":
            logger.warning(f"Empty or invalid response from web search for query: {state['search_query']}")
            return {
                "sources_gathered": [],
                "search_query": [state["search_query"]],
                "web_research_result":[]
              
                
            }
        
        # Process successful search results
        resolved_urls = resolve_urls(
            response.candidates[0].grounding_metadata.grounding_chunks, 
            state["id"], 
            disable_github_search=True
        )
        citations = get_citations(response, resolved_urls, disable_github_search=True)
        modified_text = insert_citation_markers(response.text, citations)
        sources_gathered = [item for citation in citations for item in citation["segments"]]

        logger.debug(f"Web Research Result: {response.text}")
        logger.info(f"Web research successful for query: {state['search_query']}")
        
        return {
            "sources_gathered": sources_gathered,
            "search_query": [state["search_query"]],
            "web_research_result": [response.text],
            
        }
        
    except Exception as e:
        logger.error(f"Error during web search for query '{state['search_query']}': {str(e)}")
        return {
            "sources_gathered": [],
            "search_query": [state["search_query"]],
            "web_research_result": [],
        }

def reflection(state: OverallState, config: RunnableConfig):
    """LangGraph node that reflects on the research process and results.

    This node is currently not implemented but can be used to analyze the research
    process and results for future improvements.

    Args:
        state: Current graph state containing the research results
        config: Configuration for the runnable, including LLM provider settings

    Returns:
        Dictionary with state update, including reflection key containing the reflection text
    """

    logger = logging.getLogger(f"task_{state.get('task_id', 'unknown')}")
    logger.info("Starting reflection...")
    
    configurable = Configuration.from_runnable_config(config)
    llm = ChatGoogleGenerativeAI(
        model=configurable.reflection_model,
        temperature=0.1,
        max_retries=2,
        api_key=os.getenv("GEMINI_API_KEY"),
    )
    print(f"*************************************")
    print(f"Using reflection model: {configurable.reflection_model}")
    
    info_retention = state.get("info_retention", 0)

    content_source = ""
    
    if state.get("corrected_answer"):
        content_to_reflect = state.get("corrected_answer", "")
        content_source = "corrected_answer"
        logger.info("Using corrected_answer for reflection")
    
    elif state.get("compressed_answer"):
        content_to_reflect = state.get("compressed_answer", "")
        content_source = "compressed_answer"
        logger.info("Using compressed_answer for reflection")
    else:
        content_to_reflect = state.get("finalized_answer", "")
        content_source = "finalized_answer"
        logger.info("Using finalized_answer for reflection")

    logger.info(f"Content source: {content_source}, Length: {len(content_to_reflect)}")
    state["content_to_reflect"] = content_to_reflect
    

    logger.info("Starting Stage A: Pure atomic decomposition...")
    mindtraced_research_topic = state.get("mindtraced_research_topic", "")
    
    formatted_prompt_REFLECTION_A = Reflection_A_Instructions.format(
        research_topic=get_research_topic(state.get("messages", [])),
        mindtraced_research_topic=mindtraced_research_topic,
        compressed_answer=content_to_reflect
    )

    structured_llm = llm.with_structured_output(Reflection_A)

    result = structured_llm.invoke(formatted_prompt_REFLECTION_A)
    
    task_id = state.get("task_id", "unknown")
    task_dir = state.get("task_dir", "")
    atomic_decomposition = {
        "methodology": result.feedback.atomic_information_decomposition,
        "mindtraced_atomic_units": result.feedback.information_1_atomic_sets_X,
        "spec_atomic_units": result.feedback.information_2_atomic_sets_Y,
        
    }
    
    
    logger.info(f"Decomposition complete: {len(result.feedback.information_1_atomic_sets_X)} mindtraced units, "
                f"{len(result.feedback.information_2_atomic_sets_Y)} spec units")
    
    state.update({ 
        "atomic_sets_X": result.feedback.information_1_atomic_sets_X,
        "atomic_sets_Y": result.feedback.information_2_atomic_sets_Y,
        "decomposition_method": result.feedback.atomic_information_decomposition,
        "content_to_reflect": content_to_reflect,
        "mindtraced_research_topic": mindtraced_research_topic,
        "messages": state.get("messages", []),
        "circuit_type": state.get("circuit_type", "sequential"),
        "task_id": state.get("task_id", ""),
        "task_dir": task_dir,
        

        
        "conflicts": None,  
        "reflection_feedback": None,  
        "reflectionB_feedback": None,  
       
    })
    logger.info("Reflection stage A ends, decomposition completed")
    logger.info("Starting Reflection stage B: Conflict analysis...")

    retry_reflection_count = state.get("retry_reflection_count", 0)

    atomic_sets_X = state.get("atomic_sets_X", [])
    atomic_sets_Y = state.get("atomic_sets_Y", [])
    
    logger.info(f"Analyzing conflicts between {len(atomic_sets_X)} and {len(atomic_sets_Y)} atomic units")
    
    
    formatted_prompt = Reflection_B_Instructions.format(
        research_topic=get_research_topic(state.get("messages", [])),
        mindtraced_research_topic=state.get("mindtraced_research_topic", ""),
        compressed_answer=state.get("content_to_reflect", ""),
        atomic_sets_X="\n".join(f"X{i+1}: {unit}" for i, unit in enumerate(atomic_sets_X)),
        atomic_sets_Y="\n".join(f"Y{i+1}: {unit}" for i, unit in enumerate(atomic_sets_Y))
    )
    
    structured_llm = llm.with_structured_output(Reflection_B)
    result = structured_llm.invoke(formatted_prompt) 
    intersection = result.intersection if result.intersection else []
    union = result.union if result.union else []
    logger.info(f"Reflection B result - intersection: {len(intersection)}, union: {len(union)}")
    print(f"**************************************")
    print(f" {result}")

    jaccard_similarity_value = len(intersection) / len(union) if union and len(union) > 0 else 0.0
    
    logger.info(f"Calculated Jaccard similarity: {jaccard_similarity_value:.4f}")
    jaccard_similarity_value = min(1.0, max(0.0, jaccard_similarity_value))
    state["jaccard_similarity_value"] = jaccard_similarity_value


    
    
    state.update({
    "reflectionB_feedback": result.reflectionB_feedback or result.feedback,
    "is_consistent_B": result.is_consistent_B,
    "conflicts": result.conflicts,
    "intersection": intersection,
    "union": union,
    "jaccard_similarity_value": jaccard_similarity_value
    })

   
    task_dir = state.get("task_dir", "")
    if task_dir:
        write_log_file(task_dir, "atomic_decomposition.json", 
                      json.dumps(atomic_decomposition, indent=2))
         
        units_info = {
            "timestamp": datetime.datetime.now().isoformat(),
            "jaccard_similarity": jaccard_similarity_value,
            "intersection_count": len(intersection),
            "union_count": len(union),
            "intersection_items": intersection,
            "union_items": union,
            "calculation_method": len(intersection) / len(union) if len(union) > 0 else 0,
            "is_sufficient": jaccard_similarity_value >= 0.3
        }
        write_log_file(task_dir, "units_similarity.md", 
                     f"# Units Similarity Analysis\n\n"
                     f"- Timestamp: {units_info['timestamp']}\n"
                     f"- reflectionB: {result}"
                     f"- Units Info: {units_info}\n"
                     )

 
        
    return{
       
        "jaccard_similarity_value": jaccard_similarity_value,
         "reflectionB_feedback": state.get("reflectionB_feedback", ""),
        "finalized_answer": content_to_reflect,
        "content_to_reflect": content_to_reflect,
        "compressed_answer": content_to_reflect,
        "messages": state.get("messages", []),
        "info_retention": info_retention,
        "task_id": state.get("task_id", "unknown"),
        "task_dir": state.get("task_dir", ""),
        "circuit_type": state.get("circuit_type", "unknown"),
        "retry_reflection_count": retry_reflection_count,
        "is_consistent_B": state.get("is_consistent_B", False),
    }

def reflection_continue(state: OverallState, config: RunnableConfig):
    logger = logging.getLogger(f"task_{state.get('task_id', 'unknown')}")
    retry_reflection_count = state.get("retry_reflection_count", 0)

    info_retention = state.get("info_retention", 0)
    jaccard_similarity_value = state.get("jaccard_similarity_value", 0)
    content_to_reflect = state.get("content_to_reflect", "")

    task_dir = state.get("task_dir", "")
    if jaccard_similarity_value >= 0.3 or is_consistent_B == True:  # Adjusted threshold for better sensitivity
        if jaccard_similarity_value <= 1.0:
           logger.info("Information retention sufficient, ending process.")
           state["final_answer"] = content_to_reflect
           state["compressed_answer"] = content_to_reflect

           return END
        else:
            
            # Ensure the retry counter is passed correctly
            return Send(

                "compressed_answer": content_to_reflect,
            }
              )
    
    elif retry_reflection_count >= 3:  # Use >= 3 to enforce the retry limit
        logger.info(f"Max retries exceeded for reflection (count: {retry_reflection_count}). Ending process.")
        state["final_answer"] = content_to_reflect
        state["compressed_answer"] = content_to_reflect
        if task_dir:
            write_log_file(task_dir, "max_retries_reached.txt",

        return END
    else:
        new_retry_count = retry_reflection_count + 1
        logger.info(f"Information retention insufficient. Retry {new_retry_count}/3")
        
        # Ensure the retry counter is passed correctly
        return Send(
            "finalize_answer",

                "research_topic": get_research_topic(state.get("messages", [])),

                "CMB_logic_summary": state.get("CMB_logic_summary", ""),
                "Timing_logic_summary": state.get("Timing_logic_summary", ""),
                "aggregated_results": state.get("aggregated_results", ""),
            }
        )
